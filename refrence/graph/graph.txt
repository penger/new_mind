from fastapi import FastAPI, HTTPException, Depends
from fastapi.middleware.cors import CORSMiddleware
from sqlalchemy import create_engine, Column, String, Float, ForeignKey, DateTime
from sqlalchemy.orm import declarative_base, sessionmaker, Session
from pydantic import BaseModel
from typing import List, Optional
import datetime

# ==========================================
# 1. 数据库配置 (MySQL)
# 请替换为你的 MySQL 用户名、密码和数据库名
# ==========================================
SQLALCHEMY_DATABASE_URL = "mysql+pymysql://root:123456@127.0.0.1:3306/graph_db"

engine = create_engine(SQLALCHEMY_DATABASE_URL)
SessionLocal = sessionmaker(autocommit=False, autoflush=False, bind=engine)
Base = declarative_base()

# ==========================================
# 2. SQLAlchemy 数据模型 (对应 MySQL 表)
# ==========================================
class Graph(Base):
    __tablename__ = 'graphs'
    id = Column(String(50), primary_key=True)
    name = Column(String(100), nullable=False)
    created_at = Column(DateTime, default=datetime.datetime.utcnow)

class NodeStyle(Base):
    __tablename__ = 'node_styles'
    id = Column(String(50), primary_key=True)
    graph_id = Column(String(50), ForeignKey('graphs.id', ondelete='CASCADE'))
    name = Column(String(50), nullable=False)
    color = Column(String(20), nullable=False)
    shape = Column(String(20), nullable=False)
    opacity = Column(Float, default=1.0)

class EdgeStyle(Base):
    __tablename__ = 'edge_styles'
    id = Column(String(50), primary_key=True)
    graph_id = Column(String(50), ForeignKey('graphs.id', ondelete='CASCADE'))
    name = Column(String(50), nullable=False)
    color = Column(String(20), nullable=False)
    line_style = Column(String(20), nullable=False)

class Theme(Base):
    __tablename__ = 'themes'
    id = Column(String(50), primary_key=True)
    graph_id = Column(String(50), ForeignKey('graphs.id', ondelete='CASCADE'))
    name = Column(String(50), nullable=False)
    default_node_style_id = Column(String(50), ForeignKey('node_styles.id', ondelete='SET NULL'))
    default_edge_style_id = Column(String(50), ForeignKey('edge_styles.id', ondelete='SET NULL'))

class Node(Base):
    __tablename__ = 'nodes'
    id = Column(String(50), primary_key=True)
    graph_id = Column(String(50), ForeignKey('graphs.id', ondelete='CASCADE'))
    label = Column(String(255), nullable=False)
    size = Column(Float, default=22)
    theme_id = Column(String(50), ForeignKey('themes.id', ondelete='SET NULL'))
    node_style_id = Column(String(50), ForeignKey('node_styles.id', ondelete='SET NULL'))

class Edge(Base):
    __tablename__ = 'edges'
    id = Column(String(50), primary_key=True)
    graph_id = Column(String(50), ForeignKey('graphs.id', ondelete='CASCADE'))
    source_id = Column(String(50), ForeignKey('nodes.id', ondelete='CASCADE'))
    target_id = Column(String(50), ForeignKey('nodes.id', ondelete='CASCADE'))
    label = Column(String(255))
    width = Column(Float, default=2)
    theme_id = Column(String(50), ForeignKey('themes.id', ondelete='SET NULL'))
    edge_style_id = Column(String(50), ForeignKey('edge_styles.id', ondelete='SET NULL'))

# 自动创建表结构
Base.metadata.create_all(bind=engine)

# ==========================================
# 3. Pydantic 模型 (用于接收和返回 JSON)
# ==========================================
class GraphBaseModel(BaseModel):
    id: str; name: str
class NodeStyleModel(BaseModel):
    id: str; name: str; color: str; shape: str; opacity: float
class EdgeStyleModel(BaseModel):
    id: str; name: str; color: str; style: str
class ThemeModel(BaseModel):
    id: str; name: str; defaultNodeStyleId: Optional[str]; defaultEdgeStyleId: Optional[str]
class NodeModel(BaseModel):
    id: str; label: str; size: float; themeId: Optional[str]; nodeStyleId: Optional[str]
class EdgeModel(BaseModel):
    id: str; source: str; target: str; label: str; width: float; themeId: Optional[str]; edgeStyleId: Optional[str]

class GraphDataModel(BaseModel):
    graph_id: str
    name: str
    themes: List[ThemeModel]
    nodeStyles: List[NodeStyleModel]
    edgeStyles: List[EdgeStyleModel]
    nodes: List[NodeModel]
    links: List[EdgeModel]

# ==========================================
# 4. FastAPI 路由
# ==========================================
app = FastAPI(title="Graph Editor API")

# 允许跨域请求
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_methods=["*"],
    allow_headers=["*"],
)

def get_db():
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()


# ------------------------------------------
# A. 画布全量同步接口 (供图谱编辑器主界面使用)
# ------------------------------------------
@app.get("/api/graphs/{graph_id}/data")
def get_graph_data(graph_id: str, db: Session = Depends(get_db)):
    graph = db.query(Graph).filter(Graph.id == graph_id).first()
    if not graph:
        raise HTTPException(status_code=404, detail="Graph not found")

    themes = db.query(Theme).filter(Theme.graph_id == graph_id).all()
    node_styles = db.query(NodeStyle).filter(NodeStyle.graph_id == graph_id).all()
    edge_styles = db.query(EdgeStyle).filter(EdgeStyle.graph_id == graph_id).all()
    nodes = db.query(Node).filter(Node.graph_id == graph_id).all()
    edges = db.query(Edge).filter(Edge.graph_id == graph_id).all()

    return {
        "graph_id": graph.id,
        "name": graph.name,
        "themes": [{"id": t.id, "name": t.name, "defaultNodeStyleId": t.default_node_style_id, "defaultEdgeStyleId": t.default_edge_style_id} for t in themes],
        "nodeStyles": [{"id": ns.id, "name": ns.name, "color": ns.color, "shape": ns.shape, "opacity": ns.opacity} for ns in node_styles],
        "edgeStyles": [{"id": es.id, "name": es.name, "color": es.color, "style": es.line_style} for es in edge_styles],
        "nodes": [{"id": n.id, "label": n.label, "size": n.size, "themeId": n.theme_id, "nodeStyleId": n.node_style_id} for n in nodes],
        "links": [{"id": e.id, "source": e.source_id, "target": e.target_id, "label": e.label, "width": e.width, "themeId": e.theme_id, "edgeStyleId": e.edge_style_id} for e in edges]
    }

@app.put("/api/graphs/{graph_id}/data")
def update_graph_data(graph_id: str, data: GraphDataModel, db: Session = Depends(get_db)):
    graph = db.query(Graph).filter(Graph.id == graph_id).first()
    if not graph:
        graph = Graph(id=graph_id, name=data.name)
        db.add(graph)
        db.commit()
    else:
        graph.name = data.name

    try:
        # 1. 批量清理旧数据 (注意顺序，避免外键约束报错)
        db.query(Edge).filter(Edge.graph_id == graph_id).delete()
        db.query(Node).filter(Node.graph_id == graph_id).delete()
        db.query(Theme).filter(Theme.graph_id == graph_id).delete()
        db.query(EdgeStyle).filter(EdgeStyle.graph_id == graph_id).delete()
        db.query(NodeStyle).filter(NodeStyle.graph_id == graph_id).delete()

        # 2. 批量插入新数据
        db.add_all([NodeStyle(id=ns.id, graph_id=graph_id, name=ns.name, color=ns.color, shape=ns.shape, opacity=ns.opacity) for ns in data.nodeStyles])
        db.add_all([EdgeStyle(id=es.id, graph_id=graph_id, name=es.name, color=es.color, line_style=es.style) for es in data.edgeStyles])
        db.flush() 
        db.add_all([Theme(id=t.id, graph_id=graph_id, name=t.name, default_node_style_id=t.defaultNodeStyleId, default_edge_style_id=t.defaultEdgeStyleId) for t in data.themes])
        db.add_all([Node(id=n.id, graph_id=graph_id, label=n.label, size=n.size, theme_id=n.themeId, node_style_id=n.nodeStyleId) for n in data.nodes])
        db.flush() 
        db.add_all([Edge(id=e.id, graph_id=graph_id, source_id=e.source, target_id=e.target, label=e.label, width=e.width, theme_id=e.themeId, edge_style_id=e.edgeStyleId) for e in data.links])

        db.commit()
        return {"status": "success", "message": "Graph data saved successfully"}
    except Exception as e:
        db.rollback()
        raise HTTPException(status_code=500, detail=f"Database transaction failed: {str(e)}")

# B. 图谱基础管理接口 (Graph CRUD)
# ------------------------------------------
@app.get("/api/graphs")
def get_graphs(db: Session = Depends(get_db)):
    graphs = db.query(Graph).all()
    return [{"id": g.id, "name": g.name, "created_at": g.created_at} for g in graphs]

@app.post("/api/graphs")
def create_graph(graph: GraphBaseModel, db: Session = Depends(get_db)):
    db_graph = Graph(id=graph.id, name=graph.name)
    db.add(db_graph)
    db.commit()
    return {"message": "Graph created", "id": graph.id}

@app.delete("/api/graphs/{graph_id}")
def delete_graph(graph_id: str, db: Session = Depends(get_db)):
    graph = db.query(Graph).filter(Graph.id == graph_id).first()
    if not graph:
        raise HTTPException(status_code=404, detail="Graph not found")
    db.delete(graph) # 数据库的级联删除会自动清理该图谱下的所有样式、主题、节点和连线
    db.commit()
    return {"message": "Graph and all its related elements deleted"}
